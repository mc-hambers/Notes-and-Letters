cmake_minimum_required(VERSION 3.25)
project(NotesAndLetters VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/CommonLibSSE-NG)
include(vendor/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
add_commonlibsse_plugin(NotesAndLetters NAME "Notes and Letters" AUTHOR "Maxmc" VERSION 1.0.0 SOURCES src/Plugin.cpp)
target_include_directories(NotesAndLetters PRIVATE src)
target_precompile_headers(NotesAndLetters PRIVATE src/PCH.h)
target_compile_definitions(NotesAndLetters PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NotesAndLetters PRIVATE /W4 /permissive- /utf-8)
set_target_properties(NotesAndLetters PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/SKSE/Plugins")
enable_testing()
add_executable(NoteTests tests/NoteTests.cpp)
target_include_directories(NoteTests PRIVATE src)
add_test(NAME NoteBehavior COMMAND NoteTests)
